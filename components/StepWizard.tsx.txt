export const CurrentCoverageStep = ({ userData, updateData, onNext, onBack }: StepProps) => {
  const getHomeOptions = () => {
    const base = ['Employer/Retiree Plan', 'Private Individual Insurance', 'Travel Medical Insurance', 'None'];
    const usOptions = ['Medicare A & B', 'Medigap (Supplement)', 'Medicare Advantage'];
    const caOptions = ['Provincial Coverage', 'Extended Health Benefits (Canada)'];

    if (userData.residence === 'Canada') {
      // Prioritize Canada options, but allow US ones (e.g. US citizen resident in Canada)
      return [...caOptions, ...usOptions, ...base];
    }
    // Default / US Residence
    return [...usOptions, ...caOptions, ...base];
  };

  const homeOptions = getHomeOptions();

  const toggleHome = (opt: string) => {
    const current = userData.homeCoverage;
    updateData({ 
      homeCoverage: current.includes(opt) ? current.filter(i => i !== opt) : [...current, opt] 
    });
  };

  const toggleMex = (opt: string) => {
    const current = userData.mexicoCoverage;
    updateData({ 
      mexicoCoverage: current.includes(opt) ? current.filter(i => i !== opt) : [...current, opt] 
    });
  };

  return (
    <div className="space-y-6 animate-fade-in">
      <SectionTitle>Step 2: Current Coverage</SectionTitle>
      
      <div className="bg-white p-6 rounded-xl border border-stone-200">
        <Label>Home Country Coverage (Residence: {userData.residence})</Label>
        <p className="text-sm text-stone-500 mb-3">Select all that apply to you or your partner.</p>
        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
          {homeOptions.map(opt => (
            <label key={opt} className="flex items-start gap-3 p-3 border rounded-lg cursor-pointer hover:bg-stone-50">
              <input 
                type="checkbox" 
                checked={userData.homeCoverage.includes(opt)}
                onChange={() => toggleHome(opt)}
                className="w-5 h-5 mt-0.5 text-terra-600 rounded focus:ring-terra-500 flex-shrink-0"
              />
              <span className="text-stone-700 text-sm">{opt}</span>
            </label>
          ))}
        </div>
      </div>

      <div className="bg-white p-6 rounded-xl border border-stone-200">
        <Label>Current Mexico Coverage</Label>
        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 mt-2">
          {[
            'Seguro Salud Jalisco (State Plan)', 
            'IMSS (Public Social Security)', 
            'Mexican Private Insurance', 
            'International Expat Insurance', 
            'Travel Insurance', 
            'Pay out-of-pocket'
          ].map(opt => (
            <label key={opt} className="flex items-center gap-3 p-3 border rounded-lg cursor-pointer hover:bg-stone-50">
              <input 
                type="checkbox" 
                checked={userData.mexicoCoverage.includes(opt)}
                onChange={() => toggleMex(opt)}
                className="w-5 h-5 text-terra-600 rounded focus:ring-terra-500 flex-shrink-0"
              />
              <span className="text-stone-700 text-sm">{opt}</span>
            </label>
          ))}
        </div>
      </div>

      <div>
        <Label>Experience using medical services in Mexico?</Label>
        <textarea 
          className="w-full p-3 border border-stone-300 rounded-lg h-24 bg-white"
          placeholder="Briefly describe how you've paid in the past (optional)..."
          value={userData.mexicoHistory}
          onChange={(e) => updateData({ mexicoHistory: e.target.value })}
        />
      </div>

      <div className="flex justify-between pt-4">
        <Button secondary onClick={onBack}><ChevronLeft className="w-4 h-4" /> Back</Button>
        <Button onClick={onNext}>Next Step <ChevronRight className="w-4 h-4" /></Button>
      </div>
    </div>
  );
};
